XNE_is_first_union_member = {
    custom_trigger_tooltip = {
        tooltip = XNE_is_first_union_member_tt
        OR = {
            has_country_flag = XNE_first_union_agreed
            has_country_flag = XNE_second_union_agreed
        }
    }
}
