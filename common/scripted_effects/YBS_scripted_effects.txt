YBS_become_federal_state = {
	set_country_flag = YBS_federal_state_puppet
	ROOT = {
		set_autonomy = {
			target = PREV
			autonomous_state = autonomy_YBS_federal_state
			end_wars = no
			end_civil_wars = no
		}
	}
	load_focus_tree = {
		tree = generic_focus
		keep_completed = no
	}
	add_ideas = YBS_iberian_federation_spirit

	if = {
		limit = {
			has_country_flag = YBS_federal_army_formed
		}

		add_ideas = YBS_federal_army
	}
}

YBS_calc_volunteer_manpower = {
	every_other_country = {
		limit = {
			democratic > 0
			YBS_is_iberian_country = no
		}
		set_variable = { YBS_dem_pop_to_manpower = party_popularity@democratic }
		if = {
			limit = {
				is_major = yes
			}
			multiply_variable =  { YBS_dem_pop_to_manpower =  40000 }

		}
		else = {
			multiply_variable =  { YBS_dem_pop_to_manpower =  3000 }
		}

		add_to_variable = { ROOT.YBS_volunteer_manpower = YBS_dem_pop_to_manpower }

	}
}

YBS_increase_war_exhaustion = {
	if = {
		limit = {
			has_idea = YBS_war_exhaustion
		}
		swap_ideas = {
			remove_idea = YBS_war_exhaustion
			add_idea = YBS_war_exhaustion_2
		}
	}
	else_if = {
		limit = {
			has_idea = YBS_war_exhaustion_2
		}
		swap_ideas = {
			remove_idea = YBS_war_exhaustion_2
			add_idea = YBS_war_exhaustion_3
		}
	}
	else_if = {
		limit = {
			has_idea = YBS_war_exhaustion_3
		}
		swap_ideas = {
			remove_idea = YBS_war_exhaustion_3
			add_idea = YBS_war_exhaustion_4
		}
	}
	else_if = {
		limit = {
			has_idea = YBS_war_exhaustion_4
		}
		swap_ideas = {
			remove_idea = YBS_war_exhaustion_4
			add_idea = YBS_war_exhaustion_5
		}
	}
	else_if = {
		limit = {
			has_idea = YBS_war_exhaustion_5
		}
		add_war_support = -0.05
		transfer_units_fraction= {
			target = YBS
			size = 0.05
			stockpile_ratio = 0.15
			army_ratio = 0.05
			navy_ratio = 0.05
			air_ratio = 0.1
			keep_unit_leaders_trigger = {
				has_trait = trait_SPA_nationalist_sympathies
			}
		}
	}
}
