XHH_recalculate_modifier_effects_money = {
    # gets difference between current money and last time money was recalculated
    set_temp_variable = { XHH_money_diff = XHH_money }
    subtract_from_temp_variable = { XHH_money_diff = XHH_last_money_state }
    divide_temp_variable = { XHH_money_diff = 100 }

    # multiplies difference (for modifiers where MORE)
    set_temp_variable = { XHH_money_diff_two = XHH_money_diff }
    multiply_temp_variable = { XHH_money_diff_two = 2 }

    # turns the difference negative (for modifiers where negative)
    set_temp_variable = { XHH_money_diff_negative = XHH_money_diff }
    subtract_from_temp_variable = { XHH_money_diff_negative = XHH_money_diff_two }

    # multiplies negative difference (for modifiers where negative and where MORE)
    set_temp_variable = { XHH_money_diff_negative_two = XHH_money_diff_negative }
    multiply_temp_variable = { XHH_money_diff_negative_two = 2 }

    # sets the current money as last time, for next recalculation
    set_variable = { XHH_last_money_state = XHH_money }

    custom_effect_tooltip = XHH_modify_venice_modifier_tt

    add_to_variable = { XHH_vm_manpower = 10 tooltip = weekly_manpower_tt }
    add_to_variable = { XHH_vm_nav_def = XHH_money_diff tooltip = naval_defense_factor_tt }
    add_to_variable = { XHH_vm_conv_esc = XHH_money_diff tooltip = convoy_escort_efficiency_tt }

    add_to_variable = { XHH_vm_ppf = XHH_money_diff_negative tooltip = political_power_factor_tt }
