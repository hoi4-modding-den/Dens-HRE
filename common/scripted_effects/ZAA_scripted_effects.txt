ZAA_calculate_naval_strength = {

	set_variable = { ZAA_submarine_score = num_ships_with_type@submarine }
	set_variable = { ZAA_destroyer_score = num_ships_with_type@destroyer }
	set_variable = { ZAA_light_cruiser_count = num_ships_with_type@screen_ship }
	set_variable = { ZAA_battleship_count = num_ships_with_type@capital_ship }
	set_variable = { ZAA_carrier_count = num_ships_with_type@carrier }
	
	#score calculations
	set_variable = { ZAA_light_cruiser_score = ZAA_light_cruiser_count }
	multiply_variable = { ZAA_light_cruiser_score = 3 }
	
	set_variable = { ZAA_battleship_score = ZAA_battleship_count }
	multiply_variable = { ZAA_battleship_score = 5 }
	
	set_variable = { ZAA_carrier_score = ZAA_carrier_count }
	multiply_variable = { ZAA_carrier_score = 5 }
	
	set_variable = { ZAA_naval_total_score = 0 }
	add_to_variable = { ZAA_naval_total_score = ZAA_submarine_score }
	add_to_variable = { ZAA_naval_total_score = ZAA_destroyer_score }
	add_to_variable = { ZAA_naval_total_score = ZAA_light_cruiser_score }
	add_to_variable = { ZAA_naval_total_score = ZAA_battleship_score }
	add_to_variable = { ZAA_naval_total_score = ZAA_carrier_score }

}

ZAA_calculate_loot_chance = {
	set_variable = { ZAA_loot_gain_chance = 20 }
	add_to_variable = { ZAA_loot_gain_chance = ZAA_naval_total_score }
	clamp_variable = {
		var = ZAA_loot_gain_chance
		min = 20
		max = 100
	}
	set_variable = { ZAA_nothing_ever_happens = 100 }
	subtract_from_variable = { ZAA_nothing_ever_happens = ZAA_loot_gain_chance }
	
	set_variable = { ZAA_political_power_loot_amount = ZAA_red_sea_controlled_states }
	multiply_variable = { ZAA_political_power_loot_amount = 10 }
}
