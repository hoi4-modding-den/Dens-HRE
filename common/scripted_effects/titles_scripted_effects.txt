	add_to_array = { global.HRE_title_acquire_array = token:HRE_vicounty_frankfurt }
	
	###########
	#NO TOUCHY#
	###########
	for_each_loop = {
		array = global.HRE_title_acquire_array
		value = v
		
		every_country = {
			limit = {
				any_character = {
					is_country_leader = yes
					meta_trigger = {
						text = {
							has_trait = [TRAIT]
						}
						TRAIT = "[?v.GetTokenKey]"
					}
				}
			}
			capital_scope = {
				add_to_array = { HRE_title_acquire_here = v }
				add_to_array = { global.HRE_title_states = THIS.id }
			}
		}
	}
	
	every_country = {
		capital_scope = {
			add_to_array = { HRE_title_countries = PREV.id }
		}
	}
}
