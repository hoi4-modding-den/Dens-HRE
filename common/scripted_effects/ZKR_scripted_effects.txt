ZKR_transfer_FROM_subjects_to_ROOT = {
    custom_effect_tooltip = ZKR_transfer_FROM_subjects_to_ROOT_tt
    hidden_effect = {
        every_country = {
            limit = {
                is_subject_of = FROM
            }
            ROOT = { puppet = PREV }
        }
    }
}

}

ZKR_save_current_commonwealth_status = {
    hidden_effect = {
        every_country = {
            if = {
                limit = {
                    has_idea = ZKR_pre_commonwealth_1
                }
                set_country_flag = ZKR_pre_commonwealth_1_flag
            }
            if = {
                limit = {
                    has_idea = ZKR_pre_commonwealth_2
                }
                set_country_flag = ZKR_pre_commonwealth_2_flag
            }
            if = {
                limit = {
                    has_idea = ZKR_pre_commonwealth_3
                }
                set_country_flag = ZKR_pre_commonwealth_3_flag
            }
        }
    }
}

ZKR_load_current_commonwealth_status = {
    hidden_effect = {
        every_country = {
            if = {
                limit = {
                    has_country_flag = ZKR_pre_commonwealth_1_flag
                }
                add_ideas = ZKR_pre_commonwealth_1
                clr_country_flag = ZKR_pre_commonwealth_1_flag
            }
            if = {
                limit = {
                    has_country_flag = ZKR_pre_commonwealth_2_flag
                }
                add_ideas = ZKR_pre_commonwealth_2
                clr_country_flag = ZKR_pre_commonwealth_2_flag
            }
            if = {
                limit = {
                    has_country_flag = ZKR_pre_commonwealth_3_flag
                }
                add_ideas = ZKR_pre_commonwealth_3
                clr_country_flag = ZKR_pre_commonwealth_3_flag
            }
        }
    }
}
