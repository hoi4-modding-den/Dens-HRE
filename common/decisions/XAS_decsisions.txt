	# # ### # # ### # #
	# # # # # # #   # #
	### ###  #  ### ##
	# # # #  #  #   # #
	# # # #  #  ### # #
	XAS_antitrust_laws = {
		visible = {
			check_variable = {
				var = unchecked_merchant_guilds_stab
				compare = less_than
				value = 0
			}
			check_variable = {
				var = XAS_antitrust_laws_count
				compare = less_than
				value = 3
			}
		}

		cost = 50

		priority = 1

		days_re_enable = 90
		days_remove = 30
		complete_effect = {
			add_popularity = {
				ideology = ROOT
				popularity = -0.05
			}
			custom_override_tooltip = {
				tooltip = XAS_antitrust_laws_tt
				if = {
					limit = {
						has_government = neutrality
					}
					add_timed_idea = {
						idea = XAS_antitrust_laws_neu_drift
						days = 30
					}
				}
				else_if = {
					limit = {
						has_government = democratic
					}
					add_timed_idea = {
						idea = XAS_antitrust_laws_dem_drift
						days = 30
					}
				}
				else_if = {
					limit = {
						has_government = fascism
					}
					add_timed_idea = {
						idea = XAS_antitrust_laws_fas_drift
						days = 30
					}
				}
				else = {
					add_timed_idea = {
						idea = XAS_antitrust_laws_com_drift
						days = 30
					}
				}
			}
		}
		remove_effect = {
			custom_effect_tooltip = XAS_guilds_regulation_hayek_tt
			add_to_variable = {
				var = unchecked_merchant_guilds_stab
				value = 0.01
				tooltip = stability_factor_tt
			}
			if = {
				limit = {
					check_variable = { 
						var = XAS_antitrust_laws_count
						value = 2
					}
				}
				add_to_variable = {
					var = unchecked_merchant_guilds_cg
					value = 0.05
					tooltip = consumer_goods_factor_tt
				}
			}
			hidden_effect = {
				add_to_variable = {
					XAS_antitrust_laws_count = 1
				}
			}
		}
	}
}

	# #  #   #  #   #   # #
	###  #   #  #   ### ##
	# #  #   #  #   #   # #
	# # ###  #  ### ### # #
	XAS_state_scoped_nationalization = {
		state_target = yes
		target_array = core_states
		on_map_mode = map_and_decisions_view
		target_trigger = {
			FROM = {
				can_construct_building = arms_factory
				check_variable = {
					var = XAS_nationalized_arms_industry_in_state
					value = 4
					compare = less_than_or_equals
				}
			}
		}
		target_root_trigger = {
			has_completed_focus = XAS_assume_control
		}

		cost = 25

		days_remove = 15

		remove_effect = {
			FROM = {
				if = {
					limit = {
						free_building_slots = {
							building = arms_factory
							size > 1
							include_locked = yes
						}
					}
					random_list = {
						50 = {
							add_extra_state_shared_building_slots = 1
							add_building_construction = {
								type = arms_factory
								level = 1
								instant_build = yes
							}
						}
						50 = {
							add_extra_state_shared_building_slots = 2
							add_building_construction = {
								type = arms_factory
								level = 2
								instant_build = yes
							}
						}
					}
				}
				else = {
					add_extra_state_shared_building_slots = 1
					add_building_construction = {
						type = arms_factory
						level = 1
						instant_build = yes
					}
				}
			}
			hidden_effect = { add_to_variable = { XAS_nationalized_arms_industry_in_state = 1 } }
		}
	}
