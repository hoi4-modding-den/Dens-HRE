on_actions = {
    on_startup = {
        effect = {
            YCA = {
                # AI MAIN PATH
                if = {
                    limit = {
                        is_historical_focus_on = no
                    }
                    random_list = {
                        20 = { set_country_flag = YCA_ai_path_vanguardist }
                        10 = { set_country_flag = YCA_ai_path_socialist }
                        20 = { set_country_flag = YCA_ai_path_anarchist }
                    }
                }
                else = {
                    set_country_flag = YCA_ai_path_vanguardist
                }
            }
        }
    }

	on_state_control_changed = {
		# Guerilla Tatics Stuff
		effect = {
			if = {
				limit = {
					FROM.FROM = {
						has_dynamic_modifier = { modifier = YCA_guerrilla_tactics_modifier }
					}
					FROM = { tag = FIN }
				}
				FROM.FROM = {
					remove_dynamic_modifier = { modifier = YCA_guerrilla_tactics_modifier }
				}
				#Cost decrease for Guerrilla Tactics
				hidden_effect = {
					FROM = {
						divide_variable = { FIN_motti_decision_cost = FIN_motti_decision_cost_increase }
						set_variable = { FIN_motti_decision_cost_for_trigger = FIN_motti_decision_cost } #For trigger purposes
						subtract_from_variable = { FIN_motti_decision_cost_for_trigger = 1 } 
					}
				}
			}
		}   
    }         
}
