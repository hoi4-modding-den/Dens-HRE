on_actions = {
    on_startup = {
        effect = {
            YCA = {
                # AI MAIN PATH
                if = {
                    limit = {
                        is_historical_focus_on = no
                    }
                    random_list = {
                        20 = { set_country_flag = YCA_ai_path_vanguardist }
                        10 = { set_country_flag = YCA_ai_path_socialist }
                        20 = { set_country_flag = YCA_ai_path_anarchist }
                    }
                }
                else = {
                    set_country_flag = YCA_ai_path_vanguardist
                }
            }
        }
    }
    on_monthly_YCA = {
        effect = {
            if = {
                limit = {
                    check_variable = { YCA_generalitat_faction_progress_bar > 50 }
                    NOT = {
                        has_country_flag = YCA_moderate_tensions_flag
                    }
                }
                country_event = YCA_domestic_events.12
                set_country_flag = YCA_moderate_tensions_flag
            }
        }
        effect = {
            if = {
                limit = {
                    check_variable = { YCA_generalitat_faction_progress_bar > 70 }
                    NOT = {
                        has_country_flag = YCA_high_tensions_flag
                    }
                }
                country_event = YCA_domestic_events.13
                set_country_flag = YCA_high_tensions_flag
            }
        }
        effect = {
            if = {
                limit = {
                    check_variable = { YCA_generalitat_faction_progress_bar > 90 }
                    NOT = {
                        has_country_flag = YCA_boiling_tensions_flag
                    }
                }
                country_event = YCA_domestic_events.14
                set_country_flag = YCA_boiling_tensions_flag
            }
        }                
	}
    on_weekly_YCA = {
        effect = {
            if = {
                limit = {
                    YBS = {
                        communism > 0.5
                        has_country_flag = YCA_sent_propaganda
                    }
                    NOT = {
                        has_country_flag = YCA_basque_turned_sister
                    }
                }
                country_event = YCA_domestic_events.27
                puppet = YBS
                YBS = {
                    set_cosmetic_tag = YCA_sister_republic_basque
                }
                set_country_flag = YCA_basque_turned_sister
            }
        }
        effect = {
            if = {
                limit = {
                    YRF = {
                        communism > 0.5
                        has_country_flag = YCA_sent_propaganda
                    }
                    NOT = {
                        has_country_flag = YCA_mercia_turned_sister
                    }                    
                }
                country_event = YCA_domestic_events.28
                YRF = {
                    set_cosmetic_tag = YCA_sister_republic_mercia
                }
                set_country_flag = YCA_mercia_turned_sister
            }
        }                         
	}	    	    	
	on_state_control_changed = {
		# Guerilla Tatics Stuff
		effect = {
			if = {
				limit = {
					FROM.FROM = {
						has_dynamic_modifier = { modifier = YCA_guerrilla_tactics_modifier }
					}
					FROM = { tag = FIN }
				}
				FROM.FROM = {
					remove_dynamic_modifier = { modifier = YCA_guerrilla_tactics_modifier }
				}
				#Cost decrease for Guerrilla Tactics
				hidden_effect = {
					FROM = {
						divide_variable = { FIN_motti_decision_cost = FIN_motti_decision_cost_increase }
						set_variable = { FIN_motti_decision_cost_for_trigger = FIN_motti_decision_cost } #For trigger purposes
						subtract_from_variable = { FIN_motti_decision_cost_for_trigger = 1 } 
					}
				}
			}
		}   
    }         
}
