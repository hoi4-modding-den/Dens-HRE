on_actions = {
    on_startup = {
        effect = {
            set_variable = { YGB.factory_base = 0.05 }
        }
    }

	on_weekly_YGB = {
        effect = {
            if = {
                limit = { check_variable = { political_power > 999 } }
                set_variable = { YGB.reserves_penalty = 0.002 }
            }
            else_if = {
                limit = { check_variable = { political_power > 499 } }
                set_variable = { YGB.reserves_penalty = 0.001 }
            }
            else_if = {
                limit = { check_variable = { political_power > 349 } }
                set_variable = { YGB.reserves_penalty = 0.000 }
            }
            else_if = {
                limit = { check_variable = { political_power > 199 } }
                set_variable = { YGB.reserves_penalty = -0.001 }
            }
            else_if = {
                limit = { check_variable = { political_power > 99 } }
                set_variable = { YGB.reserves_penalty = -0.002 }
            }
            else_if = {
                limit = { check_variable = { political_power > 0 } }
                set_variable = { YGB.reserves_penalty = -0.003 }
            }
            else_if = {
                limit = { check_variable = { political_power > -101 } }
                set_variable = { YGB.reserves_penalty = -0.005 }
            }
            else_if = {
                limit = { check_variable = { political_power > -201 } }
                set_variable = { YGB.reserves_penalty = -0.01 }
            }
            set_variable = { YGB.reserves_penalty_10 = YGB.reserves_penalty }
            multiply_variable = { YGB.reserves_penalty_10 = 10 }

            if = {
                limit = {
                    check_variable = { YGB.factories_assign_counter > 0 }
                }
                subtract_from_variable = { factories_assign_counter = 1 }
            }

            if = {
                limit = {
                    check_variable = { YGB.army_assign_counter > 0 }
                }
                subtract_from_variable = { army_assign_counter = 1 }
            }

        }
    }

    on_monthly_YGB = {
        effect = {
            if = {
                limit = {
                    has_completed_focus = YGB_british_deployment
                }
                add_to_variable = { YGB.british_deployment_timer = 1 }
                if = {
                    limit = {
                        check_variable = { YGB.british_deployment_timer > 5 }
                    }
                    set_variable = { YGB.british_deployment_timer = 0 }
                    CAPITAL_SCOPE = {
                        create_unit = {
                            division = "division_template = \"British Infantry Division\" start_experience_factor = 0.15"
                            owner = ROOT
                            count = 1
                        }
                    }
                }
            }
        }
    }
}
