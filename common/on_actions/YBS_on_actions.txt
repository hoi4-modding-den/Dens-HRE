on_actions = {

    on_startup = {
        effect = {
            YBS = {
                # AI MAIN PATH
                if = {
                    limit = {
                        is_historical_focus_on = yes
                        has_game_rule = {
                            rule = YBS_ai_behavior_elections
                            option = DEFAULT
                        }
                    }
                    set_country_flag = YBS_ai_path_popular_front
                }
                else_if = {
                    limit = {
                        has_game_rule = {
                            rule = YBS_ai_behavior_elections
                            option = YBS_SOCIALIST_OPTION
                        }
                    }
                    set_country_flag = YBS_ai_path_socialist
                }
                else_if = {
                    limit = {
                        has_game_rule = {
                            rule = YBS_ai_behavior_elections
                            option = YBS_NATIONALIST_OPTION
                        }
                    }
                    set_country_flag = YBS_ai_path_nationalist
                }
                else_if = {
                    limit = {
                        has_game_rule = {
                            rule = YBS_ai_behavior_elections
                            option = YBS_LIBERAL_OPTION
                        }
                    }
                    set_country_flag = YBS_ai_path_liberals
                }
                else_if = {
                    limit = {
                        has_game_rule = {
                            rule = YBS_ai_behavior_elections
                            option = YBS_POPULAR_FRONT_OPTION
                        }
                    }
                    set_country_flag = YBS_ai_path_popular_front
                }
                else_if = {
                    limit = {
                        has_game_rule = {
                            rule = YBS_ai_behavior_elections
                            option = YBS_CARLIST_OPTION
                        }
                    }
                    set_country_flag = YBS_ai_path_carlist
                }
                else_if = {
                    limit = {
                        has_game_rule = {
                            rule = YBS_ai_behavior_elections
                            option = YBS_SYNDICALIST_OPTION
                        }
                    }
                    set_country_flag = YBS_ai_path_syndicalist
                }
                else = {
                    random_list = {
                        10 = { set_country_flag = YBS_ai_path_carlist }
                        10 = { set_country_flag = YBS_ai_path_nationalist }
                        20 = { set_country_flag = YBS_ai_path_liberals }
                        10 = { set_country_flag = YBS_ai_path_socialist }
                        20 = { set_country_flag = YBS_ai_path_syndicalist }
                        30 = { set_country_flag = YBS_ai_path_popular_front }
                    }
                }


                # AI DEMOCRATIC SUBPATH (useless if not democratic)
                if = {
                    limit = {
                        is_historical_focus_on = yes
                        has_game_rule = {
                            rule = YBS_ai_behavior_democratic_path
                            option = DEFAULT
                        }
                    }
                    set_country_flag = YBS_ai_path_federation
                }
                else_if = {
                    limit = {
                        has_country_flag = YBS_ai_path_liberals
                        has_game_rule = {
                            rule = YBS_ai_behavior_democratic_path
                            option = YBS_DEMOCRATIC_CORPORATION_OPTION
                        }
                    }
                    set_country_flag = YBS_ai_path_capitalism
                }
                else_if = {
                    limit = {
                        has_game_rule = {
                            rule = YBS_ai_behavior_democratic_path
                            option = YBS_DEMOCRATIC_FEDERATION_OPTION
                        }
                    }
                    set_country_flag = YBS_ai_path_federation
                }
                else_if = {
                    limit = {
                        has_game_rule = {
                            rule = YBS_ai_behavior_democratic_path
                            option = YBS_DEMOCRATIC_REPUBLIC_OPTION
                        }
                    }
                    set_country_flag = YBS_ai_path_republic
                }
                else_if = {
                    limit = {
                        has_country_flag = YBS_ai_path_popular_front
                        has_game_rule = {
                            rule = YBS_ai_behavior_democratic_path
                            option = YBS_DEMOCRATIC_ESPERANTO_OPTION
                        }
                    }
                    set_country_flag = YBS_ai_path_esperantism
                }
                else = {
                    random_list = {
                        40 = { modifier = { factor = 0 NOT = { has_country_flag = YBS_ai_path_liberals } } set_country_flag = YBS_ai_path_capitalism }
                        10 = { set_country_flag = YBS_ai_path_federation }
                        10 = { set_country_flag = YBS_ai_path_republic }
                        40 = { modifier = { factor = 0 NOT = { has_country_flag = YBS_ai_path_popular_front } } set_country_flag = YBS_ai_path_esperantism }
                    }
                }


                # AI CARLIST SUBPATH (useless if not carlist)
                if = {
                    limit = {
                        has_game_rule = {
                            rule = YBS_ai_behavior_carlist_path
                            option = YBS_CARLIST_HONOUR_OPTION
                        }
                    }
                    set_country_flag = YBS_ai_path_honour
                }
                else_if = {
                    limit = {
                        has_game_rule = {
                            rule = YBS_ai_behavior_carlist_path
                            option = YBS_CARLIST_TRICKSTER_OPTION
                        }
                    }
                    set_country_flag = YBS_ai_path_trickster
                }
                else = {
                    random_list = {
                        70 = { set_country_flag = YBS_ai_path_honour }
                        30 = { set_country_flag = YBS_ai_path_trickster }
                    }
                }


                if = {
                    limit = {
                        YBS = { is_ai = yes }
                    }
                    add_ideas = YBS_ai_army_assistance # because AI is stupid
                }
            }
        }
    }

    on_ruling_party_change = {
        effect = {
            if = {
                limit = {
                    original_tag = YBS
                }

                # Communism
                if = {
                    limit = {
                        has_government = communism
                    }
                    if = {
                        limit = {
                            has_country_flag = YBS_united_iberia
                        }
                        set_cosmetic_tag = YBS_iberian_defence_council
                    }
                    else_if = {
                        limit = {
                            NOT = { has_cosmetic_tag = YBS_unifier }
                        }
                        set_cosmetic_tag = YBS_syndicalist
                    }
                }

                # Democratic
                if = {
                    limit = {
                        has_government = democratic
                    }
                    if = {
                        limit = {
                            has_country_flag = YBS_united_iberia
                        }
                        set_cosmetic_tag = YBS_iberia
                    }
                    else_if = {
                        limit = {
                            has_country_flag = YBS_rde_flag_1
                        }
                        set_cosmetic_tag = YBS_esperanto_republic_1
                    }
                    else_if = {
                        limit = {
                            has_country_flag = YBS_rde_flag_2
                        }
                        set_cosmetic_tag = YBS_esperanto_republic_2
                    }
                    else_if = {
                        limit = {
                            has_country_flag = YBS_rde_flag_3
                        }
                        set_cosmetic_tag = YBS_esperanto_republic_3
                    }
                    else_if = {
                        limit = {
                            has_country_flag = YBS_federal_state_formed
                        }
                        set_cosmetic_tag = YBS_federal_state
                    }
                }

                # Neutrality
                if = {
                    limit = {
                        has_government = neutrality
                    }
                    if = {
                        limit = {
                            has_country_flag = YBS_hanse_takeover
                        }
                        if = {
                            limit = {
                                has_country_flag = YBS_united_iberia
                            }
                            set_cosmetic_tag = YBS_hanse_iberia
                        }
                        else = {
                            set_cosmetic_tag = YBS_hanse
                        }
                    }
                    else = {
                        if = {
                            limit = {
                                has_country_flag = YBS_united_iberia
                            }
                            set_cosmetic_tag = YBS_iberia_carlist
                        }
                    }
                }

                # Fascism
                if = {
                    limit = {
                        has_government = fascism
                    }
                    if = {
                        limit = {
                            has_country_flag = YBS_united_iberia
                        }
                        set_cosmetic_tag = YBS_iberia_falangist
                    }
                }

                # Remove Checks
                if = {
                    limit = {
                        NOT = { has_government = communism }
                        has_cosmetic_tag = YBS_syndicalist
                    }
                    drop_cosmetic_tag = yes
                }
                if = {
                    limit = {
                        NOT = { has_government = neutrality }
                        has_cosmetic_tag = YBS_hanse
                    }
                    drop_cosmetic_tag = yes
                }
                if = {
                    limit = {
                        NOT = { has_government = democratic }
                        has_cosmetic_tag = YBS_federal_state
                    }
                    drop_cosmetic_tag = yes
                }
                if = {
                    limit = {
                        OR = {
                            has_cosmetic_tag = YBS_esperanto_republic_1
                            has_cosmetic_tag = YBS_esperanto_republic_2
                            has_cosmetic_tag = YBS_esperanto_republic_3
                        }
                        NOT = { has_government = democratic }
                    }
                    drop_cosmetic_tag = yes
                }

                # UNIFIER Tag
                if = {
                    limit = {
                        has_government = neutrality
                        OR = {
                            has_cosmetic_tag = YBS_unifier
                            has_country_flag = YBS_became_unifier
                        }
                    }
                    set_cosmetic_tag = YBS_unifier_carlist
                }
                if = {
                    limit = {
                        NOT = { has_government = neutrality }
                        OR = {
                            has_cosmetic_tag = YBS_unifier_carlist
                            has_country_flag = YBS_became_unifier
                        }
                    }
                    set_cosmetic_tag = YBS_unifier
                }
            }
        }
    }

    on_war_relation_added = {
        effect = {
            if = {
                limit = {
                    ROOT = { original_tag = YZA }
                    FROM = { original_tag = YBS has_completed_focus = YBS_basque_military_academy }
                }
                FROM = { country_event = YBS_diplo_events.29 }
            }
            if = {
                limit = {
                    ROOT = { original_tag = YAO }
                    FROM = { original_tag = YBS has_completed_focus = YBS_basque_military_academy }
                }
                FROM = { country_event = YBS_diplo_events.30 }
            }
            if = {
                limit = {
                    ROOT = { original_tag = YCA }
                    FROM = { original_tag = YBS has_completed_focus = YBS_basque_military_academy }
                }
                FROM = { country_event = YBS_diplo_events.31 }
            }
        }
    }

}
