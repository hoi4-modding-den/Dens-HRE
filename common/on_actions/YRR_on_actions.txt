on_actions = {

    on_ruling_party_change = {
        effect = {
            if = {
                limit = {
                    original_tag = YRR
                }
                if = {
                    limit = {
                        has_cosmetic_tag = YRR_spanish_pretenders
                        OR = {
                            has_government = communism
                            has_government = democratic
                        }
                    }
                    drop_cosmetic_tag = yes
                    set_country_flag = YBS_dropped_pretender_tag
                }
                if = {
                    limit = {
                        has_country_flag = YBS_dropped_pretender_tag
                        NOT = { has_cosmetic_tag = YRR_spanish_pretenders }
                        NOT = { has_cosmetic_tag = YRR_spain }
                        OR = {
                            has_government = neutrality
                            has_government = fascism
                        }
                    }
                    set_cosmetic_tag = YRR_spanish_pretenders
                }
            }
        }
    }

}
