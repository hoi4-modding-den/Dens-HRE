on_actions = {

    on_startup = {
        effect = {
            ZKR = {
                add_state_core = 353
                add_state_core = 419
                add_state_core = 1001
                add_state_core = 1103
                add_state_core = 1115
                add_state_core = 1117
                add_state_core = 676
                add_state_core = 1116

                add_state_claim = 1113
                add_state_claim = 355
                add_state_claim = 349
                add_state_claim = 798
                add_state_claim = 348
                add_state_claim = 680
                add_state_claim = 354
                add_state_claim = 344

                every_core_state = {
                    add_to_array = { ZKR_starting_cores = THIS }
                }
                every_state = {
                    limit = {
                        OR = {
                            is_core_of = ZKR
                            is_claimed_by = ZKR
                        }
                    }
                    set_state_flag = ZKR_greater_kurdistan_state
                }

                # Army Stuff

                every_army_leader = {
                    set_character_flag = ZKR_not_generate_character
                }

                every_owned_state = {
                    every_state_division = {
                        limit = {
                            NOT = { division_has_majority_template = infantry }
                        }
                        add_history_entry = {
                            key = ZKR_independence_war_history_entry
                            subject = "ZKR_independence_war_history_entry"
                            allow = no
                        }
                        random_list = {
                            75 = {
                                add_divisional_commander_xp = 150
                            }
                            25 = {
                                add_divisional_commander_xp = 300
                            }
                        }
                    }
                    random_state_division = {
                        limit = {
                            NOT = { division_has_majority_template = infantry }
                        }
                        random_list = {
                            10 = {
                                add_divisional_commander_xp = 750
                            }
                            50 = {
                                add_divisional_commander_xp = 500
                            }
                            40 = {
                                add_divisional_commander_xp = 250
                            }
                        }
                        promote_officer_to_general = yes
                    }
                }

                every_army_leader = {
                    limit = {
                        NOT = { has_character_flag = ZKR_not_generate_character }
                    }
                    add_unit_leader_trait = war_hero
                }

                every_army_leader = {
                    limit = {
                        has_character_flag = ZKR_not_generate_character
                    }
                    clr_character_flag = ZKR_not_generate_character
                }
            }
        }
    }

    on_ruling_party_change = {
        effect = {
            if = {
                limit = {
                    tag = ZKR
                }
                mark_focus_tree_layout_dirty = yes
                if = {
                    limit = {
                        has_government = neutrality
                        NOT = { has_country_leader = { character = ZKR_mahmud_barzanji ruling_only = yes } }
                    }
                    if = {
                        limit = {
                            OR = {
                                has_country_flag = ZKR_mesopotamia_formed
                                has_country_flag = ZKR_commonwealth_formed
                            }
                        }
                        set_cosmetic_tag = ZKR_british_puppet_mesopotamia
                    }
                    else = {
                        set_cosmetic_tag = ZKR_british_puppet
                    }
                }
                else = {
                    if = {
                        limit = {
                            has_cosmetic_tag = ZKR_british_puppet_mesopotamia
                        }
                        set_cosmetic_tag = ZKR_commonwealth
                    }
                    if = {
                        limit = {
                            has_cosmetic_tag = ZKR_british_puppet
                        }
                        drop_cosmetic_tag = yes
                    }
                }
                if = {
                    limit = {
                        NOT = { has_country_leader = { character = ZKR_mahmud_barzanji ruling_only = yes } }
                        has_cosmetic_tag = ZKR_hegemony_of_kurdistan
                    }
                    drop_cosmetic_tag = yes
                }
                else_if = {
                    limit = {
                        has_country_leader = { character = ZKR_mahmud_barzanji ruling_only = yes }
                        has_country_flag = ZKR_hegemony_of_kurdistan_formed
                        NOT = { has_cosmetic_tag = ZKR_hegemony_of_kurdistan }
                    }
                    set_cosmetic_tag = ZKR_hegemony_of_kurdistan
                }
            }
        }
    }

    on_send_volunteers = {
        effect = {
            if = {
                limit = {
                    ROOT = { tag = ZKR }
                    FROM = { ZKR_is_syrian_nation = yes }
                }
                FROM = { set_country_flag = ZKR_sent_us_volunteers }
            }
        }
    }

    on_capitulation = {
        effect = {
            if = {
                limit = {
                    FROM = { tag = ZKR }
                    ROOT = { original_tag = ZNW }
                }
                ROOT = {
                    every_core_state = {
                        remove_core_of = ROOT
                    }
                }
            }
            if = {
                limit = {
                    ROOT = { has_country_flag = ZKR_conquest_target }
                }
                ROOT = { clr_country_flag = ZKR_conquest_target }
            }
            if = {
                limit = {
                    FROM = { tag = ZKR }
                    NOT = {
                        any_country = {
                            NOT = { tag = FROM }
                            has_war_with = ROOT
                        }
                    }
                }
                if = {
                    limit = {
                        ROOT = {
                            all_owned_state = {
                                has_state_flag = ZKR_greater_kurdistan_state
                            }
                        }
                    }
                    FROM = {
                        white_peace = {
                            tag = ROOT
                            message = ZKR_annexed_before_conference_tt
                        }
                        annex_country = {
                            target = ROOT
                            transfer_troops = no
                        }
                    }
                }
            }
            if = {
                limit = {
                    FROM = { tag = ZKR }
                    ROOT = { has_character = ZCC_Josef_Dzhugashvili }
                }
                FROM = { country_event = { id = ZKR_lore_events.4 days = 3 } }
            }
        }
    }

    on_subject_autonomy_level_change = {
        effect = {
            if = {
                limit = {
                    ROOT = { 
                        has_country_flag = ZKR_federal_state_puppet 
                        NOT = { has_autonomy_state = autonomy_ZKR_commonwealth_associate }
                    }
                    FROM = { original_tag = ZKR }
                }
                FROM = { end_puppet = ROOT }
            }
        }
    }

    on_new_term_election = {
		random_events = {
			100 = ZKR_politics_events.4
		}
	}

    on_justifying_wargoal_pulse = {
		random_events = {
			100 = ZKR_technical_events.24
		}
	}

    on_declare_war = {
        effect = {
            if = {
                limit = {
                    ROOT = { 
                        has_completed_focus = ZKR_secure_peace
                        NOT = { has_country_flag = ZKR_peace_lore_event_triggered }
                    }
                }
                country_event = { id = ZKR_politics_events.9 days = 10 }
            }
        }
    }

}
