on_actions = {
    on_startup = {
        effect = {
            YKP = {
                if = {
                    limit = {
                        has_game_rule = {
                            rule = YKP_ai_behaviour_path
                            option = YKP_MARIA_IV_OPTION
                        }
                    }

                    set_country_flag = YKP_AI_Maria_IV_path
                }

                if = {
                    limit = {
                        has_game_rule = {
                            rule = YKP_ai_behaviour_path
                            option = YKP_MILITARY_OPTION
                        }
                    }

                    set_country_flag = YKP_AdI_Military_path
                }
            }
        }
    }

	on_weekly_YKP = {
        effect = {
            if = {
                limit = {
                    YKP = {
                        has_country_leader = {
                            character = YKP_maria_adelaide_de_braganca
                            ruling_only = yes
                        }
                    }
                }

                YKP_update_conquer_targets = yes
            }
        }
    }
}
