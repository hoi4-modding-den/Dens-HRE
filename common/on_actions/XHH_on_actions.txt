on_actions = {

    on_ruling_party_change = {
        effect = {
            if = {
                limit = {
                    original_tag = XHH
                }
                if = {
                    limit = {
                        XHH_has_monarch = yes
                    }
                    if = {
                        limit = {
                            has_government = democratic
                        }
                        XHH_tom_first = {
                            set_portraits = {
                                civilian = { large = GFX_portrait_XHH_tom_first_civilian }
                            }
                        }
                    }
                    else = {
                        XHH_tom_first = {
                            set_portraits = {
                                civilian = { large = GFX_portrait_XHH_tom_first }
                            }
                        }
                    }
                }
                if = {
                    limit = {
                        XHH_has_monarch = no
                    }
                    set_country_flag = HRE_is_republic
                    if = {
                        limit = {
                            NOT = { has_cosmetic_tag = XHH_germany }
                            NOT = { has_cosmetic_tag = XHH_north_germany }
                            NOT = { has_cosmetic_tag = XHH_trade_federation }
                        }
                        set_cosmetic_tag = XHH_republic
                    }
                }
                if = {
                    limit = {
                        has_country_flag = HRE_is_republic
                        XHH_has_monarch = yes
                    }
                    clr_country_flag = HRE_is_republic
                    if = {
                        limit = {
                            has_cosmetic_tag = XHH_republic
                        }
                        drop_cosmetic_tag = yes
                    }
                }
                if = {
                    limit = {
                        OR = {
                            has_government = fascism
                            AND = {
                                has_government = neutrality
                                has_country_flag = XHH_senate_removed
                            }
                        }
                    }
                    set_variable = { HRE_national_stance_welfare = 3 }
                }
                if = {
                    limit = {
                        OR = {
                            has_government = communism
                            has_government = democratic
                            AND = {
                                has_government = neutrality
                                NOT = { has_country_flag = XHH_senate_removed }
                            }
                        }
                    }
                    set_variable = { HRE_national_stance_welfare = 1 }
                }
            }
            if = {
                limit = {
                    original_tag = XH1
                }
                if = {
                    limit = {
                        NOT = {
                            has_country_leader = { character = XH1_ernest_august ruling_only = yes }
                        }
                    }
                    set_country_flag = HRE_is_republic
                }
                if = {
                    limit = {
                        has_country_flag = HRE_is_republic
                        has_country_leader = { character = XH1_ernest_august ruling_only = yes }
                    }
                    clr_country_flag = HRE_is_republic
                }
            }
            if = {
                limit = {
                    original_tag = XH2
                }
                if = {
                    limit = {
                        NOT = {
                            has_country_leader = { character = XH2_christian_louis ruling_only = yes }
                        }
                    }
                    set_country_flag = HRE_is_republic
                }
                if = {
                    limit = {
                        has_country_flag = HRE_is_republic
                        has_country_leader = { character = XH2_christian_louis ruling_only = yes }
                    }
                    clr_country_flag = HRE_is_republic
                }
            }
        }
    }

    on_release_as_free = {
        effect = {
            if = {
                limit = {
                    ROOT = {
                        OR = {
                            original_tag = XH1
                            original_tag = XH2
                        }
                        has_government = neutrality
                    }
                }
                add_to_array = { global.HRE_members = THIS.id }
				add_to_array = { global.HRE_voting_array = THIS.id }
				set_country_flag = HRE_member_state
				add_dynamic_modifier = { modifier = HRE_member_state_spirit }
            }
        }
    }
    on_release_as_puppet = {
        effect = {
            if = {
                limit = {
                    ROOT = {
                        OR = {
                            original_tag = XH1
                            original_tag = XH2
                        }
                        has_government = neutrality
                    }
                }
                add_to_array = { global.HRE_members = THIS.id }
				add_to_array = { global.HRE_voting_array = THIS.id }
				set_country_flag = HRE_member_state
				add_dynamic_modifier = { modifier = HRE_member_state_spirit }
            }
        }
    }

    on_capitulation = {
        effect = {
            if = {
                limit = {
                    NOT = {
                        any_country = {
                            NOT = { tag = FROM }
                            has_war_with = ROOT
                        }
                    }
                    ROOT = {
                        any_owned_state = {
                            is_claimed_by = FROM
                            is_controlled_by = FROM
                        }
                    }
					FROM = { 
                        original_tag = XHH
                        is_ai = yes
                    }
                }
                ROOT = {
                    every_owned_state = {
                        limit = {
                            is_claimed_by = FROM
                            is_controlled_by = FROM
                        }
                        transfer_state_to = FROM
                    }
                }
            }
        }
    }

    on_state_control_changed = {
        effect = {
            if = {
                limit = {
                    ROOT = { original_tag = XHH }
                }
                FROM.FROM = {
                    every_neighbor_state = {
                        limit = {
                            is_owned_by = ZZZ
                            has_state_flag = XHH_seize_state_from_ZZZ
                        }
                        state_event = {
                            id = XHH_state_seized_event.1
                            days = 10
                            random_days = 5
                            trigger_for = owner
                        }
                    }
                }
            }
        }
    }

    on_new_term_election = {
		random_events = {
			100 = XHH_internal_events.6
		}
	}
}
