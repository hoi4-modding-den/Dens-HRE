country_event = {
	id = XNE_joint.1
	title = XNE_joint.1.t
	desc = XNE_joint.1.desc
	
	picture = GFX_report_event_generic_conference
	timeout_days = 4
	
	is_triggered_only = yes
	
	option = {
        name = XNE_joint.1.a
		trigger = {
			NOT = {
				OR = {
					has_idea = XHO_calvinists_bolstered_1
					has_idea = XHO_calvinists_bolstered_2
					has_idea = XHO_calvinists_bolstered_3
				}
			}
		}
		ai_chance = {
			factor = 10
			modifier = {
				tag = XHO
				factor = 0
			}
		}
		set_country_flag = XNE_first_union_agreed
		custom_effect_tooltip = XNE_joint.1.a.tt1
		if = {
			limit = {
				has_idea = XHO_calvinist_problem
			}
			custom_effect_tooltip = XNE_joint.1.a.tt2
			add_stability = -0.1
			add_political_power = -100
			start_civil_war = {
				ideology = fascism
				size = 0.4
			}
			hidden_effect = {
				if = {
					limit = {
						has_completed_focus = XHO_power_to_the_monarch
					}
					complete_national_focus = XHO_root_out_dissenters
				}
				else_if = {
					limit = {
						has_completed_focus = XHO_reaffirm_religious_freedoms
					}
					if = {
						limit = {
							NOT = { has_completed_focus = XHO_liberal_movement }
						}
						unlock_national_focus = XHO_liberal_movement
					}
					complete_national_focus = XHO_calvanist_concessions
				}
				else_if = {
					limit = {
						has_completed_focus = XHO_crack_down_on_calvinism
					}
					if = {
						limit = {
							NOT = { has_completed_focus = XHO_counterespionage }
						}
						unlock_national_focus = XHO_counterespionage
					}
					complete_national_focus = XHO_power_to_the_monarch
					complete_national_focus = XHO_root_out_dissenters
				}
				else = {
					unlock_national_focus = XHO_crack_down_on_calvinism
					unlock_national_focus = XHO_counterespionage
					complete_national_focus = XHO_power_to_the_monarch
					complete_national_focus = XHO_root_out_dissenters
				}
			}
		}
    }

    option = {
        name = XNE_joint.1.b
		ai_chance = {
			factor = 1
			modifier = {
				tag = XHO
				factor = 10
			}
		}
		custom_effect_tooltip = XNE_joint.1.b.tt
		if = {
			limit = {
				NOT = { tag = XHO }
			}
			add_political_power = -10
		}
		else = {
			add_political_power = 25
		}
    }
}

country_event = {
	id = XNE_joint.4
	title = XNE_joint.4.t
	desc = XNE_joint.4.d
	
	picture = GFX_report_event_generic_conference
	
	is_triggered_only = yes

	option = {
        name = XNE_joint.4.a
		trigger = {
			any_other_country = {
				has_country_flag = XNE_second_union_agreed
			}
		}
		meta_effect = {
			text = {
				create_faction = "[X]"
			}
			X = "[GetUnionName]"
		}

		if = {
			limit = {
				NOT = {
					has_dynamic_modifier = {
						modifier = XNE_joint_idea
					}
				}
			}
			add_dynamic_modifier = {
				modifier = XNE_joint_idea
			}
		}

		every_other_country = {
			limit = {
				has_country_flag = XNE_second_union_agreed
				NOT = { is_in_faction = yes }
			}
			PREV = { add_to_faction = PREV }
				if = {
					limit = {
						NOT = {
							has_dynamic_modifier = {
								modifier = XNE_joint_idea
							}
						}
					}
					add_dynamic_modifier = {
						modifier = XNE_joint_idea
					}
				}
		}
		custom_effect_tooltip = XNE_joint_second_union_complete_tt
		set_variable = { XNE.union_capital = var:capital }
		add_to_variable = { XNE.joint_pp = 0.1 }
	}

	option = {
        name = XNE_joint.4.b
		trigger = {
			all_other_country = {
				NOT = { has_country_flag = XNE_second_union_agreed }
			}
		}
		add_stability = -0.1
		add_political_power = -50
		clr_country_flag = XNE_second_union_agreed
		set_variable = { XNE.union_capital = -1 }
	}
}
