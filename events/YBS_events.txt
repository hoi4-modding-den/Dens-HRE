country_event = { # Election Week begins
	id = YBS_election_events.7
	title = YBS_election_events.7.t
	desc = YBS_election_events.7.desc
    picture = GFX_report_event_communists_cheer
	
	is_triggered_only = yes
	
	option = {
        name = YBS_election_events.7.a
        activate_shine_on_focus = YBS_election_week
        set_country_flag = YBS_elections_happened

        hidden_effect = {
            if = {
				limit = {
					OR = {
						AND = {
							check_variable = { YBS_con_pop > YBS_mon_pop }
							check_variable = { YBS_con_pop > YBS_lib_pop }
							check_variable = { YBS_con_pop > YBS_soc_pop }
							check_variable = { YBS_con_pop > YBS_syn_pop }
							check_variable = { YBS_con_pop > YBS_pf_pop } 
						}
						NOT = { has_country_flag = YBS_carlist_compromise_made }
					}
					NOT = { has_country_flag = YBS_carlist_coup_stopped }
				}
				country_event = { id = YBS_election_events.5 days = 7 }
			}
			else_if = {
				limit = {
					check_variable = { YBS_mon_pop > YBS_con_pop }
					check_variable = { YBS_mon_pop > YBS_lib_pop }
					check_variable = { YBS_mon_pop > YBS_soc_pop }
					check_variable = { YBS_mon_pop > YBS_syn_pop }
					check_variable = { YBS_mon_pop > YBS_pf_pop }
				}
				country_event = { id = YBS_election_events.6 days = 7 }
			}
			else = {
				country_event = { id = YBS_election_events.2 days = 7 }
			}
        }
    }
}
